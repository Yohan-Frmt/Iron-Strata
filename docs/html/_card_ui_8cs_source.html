120 public override void _GuiInput(InputEvent @event) {
121 Main main = GetTree().Root.GetNodeOrNull<Main>("Main");
122
123 switch (@event) {
124 case InputEventMouseButton { ButtonIndex: MouseButton.Left } mouseButton:
125 if (mouseButton.Pressed) {
126 if (GetCurrentScrap() < PlayCost) {
127 return;
128 }
129
130 _isDragging = true;
131 IsAnyCardDragged = true;
132 _startPosition = GlobalPosition;
133 TopLevel = true;
134 ZIndex = 100;
135 Modulate = new Color(1f, 1f, 1f, 0.4f);
136 }
137 else if (_isDragging) {
138 _isDragging = false;
139 IsAnyCardDragged = false;
140 ZIndex = 0;
141 Modulate = new Color(1f, 1f, 1f);
142 main?.HidePreview();
143
144 bool success = main != null && main.TryPlayCard(TypeToApply, PlayCost, GetGlobalMousePosition()).IsOk;
145 if (!success) {
146 TopLevel = false;
147 GlobalPosition = _startPosition;
148 }
149 else {
150 QueueFree();
151 }
152 }
153 break;
154
155 case InputEventMouseMotion mouseMotion when _isDragging:
156 GlobalPosition += mouseMotion.Relative;
157 main?.UpdatePreview(TypeToApply, GetGlobalMousePosition());
158 break;
159 }
