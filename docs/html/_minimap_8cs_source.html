245 private static Vector2 GetTrainMapPosition(MapComponent map, LocationComponent location) {
246 Vector2 start = map.AllNodes[location.CurrentNodeId].Position;
247 Vector2 end = map.AllNodes[location.TargetNodeId].Position;
248 Vector3 startPosition3D = new(start.X, 0, start.Y);
249 Vector3 endPosition3D = new(end.X, 0, end.Y);
250 float segmentDistance = startPosition3D.DistanceTo(endPosition3D);
251 float interpolationFactor = segmentDistance > 0 ? Mathf.Clamp(location.TravelProgress / segmentDistance, 0, 1) : 0;
252 Vector3 position3D = RailSampler.SampleBezier(startPosition3D, endPosition3D, location.CurrentNodeId, location.TargetNodeId, interpolationFactor);
253 return new Vector2(position3D.X, position3D.Z);
254 }
