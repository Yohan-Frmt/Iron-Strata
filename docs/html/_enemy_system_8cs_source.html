29 public void Update(World world, double delta) {
30 Option<Entity> locationEntityOption = world.QueryFirst<LocationComponent>();
31 bool isInCity = false;
32 if (locationEntityOption.IsSome) { isInCity = world.Get<LocationComponent>(locationEntityOption.Unwrap()).IsInCityZone; }
33
34 if (isInCity) {
35 _hordeTimer += (float)delta;
36 if (_hordeTimer >= _hordeSpawnInterval) {
37 _hordeTimer = 0f;
38 foreach (HordeSpawnRule rule in EnemyRegistry.SpawnRules) {
39 if (GD.Randf() <= rule.Chance) { SpawnHorde(world, rule.Count, rule.Type); }
40 }
41 }
42 }
43 else { _hordeTimer = 0f; }
44
45 List<Entity> wagons = [.. world.Query<WagonSlotComponent, WagonTypeComponent, HealthComponent>()];
46 if (wagons.Count == 0) { return; }
47
48 List<Entity> allEnemies = [.. world.Query<EnemyComponent, PositionComponent>()];
49
50 foreach (Entity entity in world.Query<EnemyComponent, PositionComponent, MovementComponent>()) {
51 ref EnemyComponent enemy = ref world.Get<EnemyComponent>(entity);
52 ref PositionComponent positionComponent = ref world.Get<PositionComponent>(entity);
53 ref MovementComponent movementComponent = ref world.Get<MovementComponent>(entity);
54 bool needsNewTarget = enemy.CurrentTarget.Match(target => !world.IsAlive(target), () => true);
55 if (needsNewTarget) { enemy.CurrentTarget = FindBestTarget(enemy.Type, wagons, world); }
56
57 if (enemy.CurrentTarget.IsNone) { continue; }
58
59 Entity target = enemy.CurrentTarget.Unwrap();
60 ref WagonSlotComponent wagonSlotComponent = ref world.Get<WagonSlotComponent>(target);
61 const float wagonSize = 5f;
62 const float wagonPhysicalRadius = 4.0f;
63 Vector3 targetGlobalPosition = trainRoot.GlobalPosition + new Vector3(-wagonSlotComponent.SlotIndex * wagonSize, 2.0f, 0);
64 float distanceToTargetCenter = positionComponent.Value.DistanceTo(targetGlobalPosition);
65 if (distanceToTargetCenter <= enemy.AttackRange + wagonPhysicalRadius) {
66 enemy.AttackTimer -= (float)delta;
67 if (!(enemy.AttackTimer <= 0)) { continue; }
68
69 enemy.AttackTimer = 1f / enemy.AttackSpeed;
70 ref HealthComponent health = ref world.Get<HealthComponent>(target);
71 health.Current -= enemy.Damage;
72 if (health.Current <= 0) { HandleWagonDestruction(world, target); }
73 }
74 else {
75 Vector3 moveDirection = (targetGlobalPosition - positionComponent.Value).Normalized();
76 Vector3 repulsionForce = Vector3.Zero;
77 int neighbors = 0;
78 foreach (Entity otherEntity in allEnemies) {
79 if (entity.Equals(otherEntity)) { continue; }
80
81 Vector3 otherPosition = world.Get<PositionComponent>(otherEntity).Value;
82 float distanceToOtherEnemy = positionComponent.Value.DistanceTo(otherPosition);
83 const float separationRadius = 10.0f;
84 if (!(distanceToOtherEnemy < separationRadius) || !(distanceToOtherEnemy > 0.01f)) { continue; }
85
86 Vector3 pushDirection = (positionComponent.Value - otherPosition).Normalized();
87 repulsionForce += pushDirection * (separationRadius - distanceToOtherEnemy);
88 neighbors++;
89 }
90
91 foreach (Entity wagon in wagons) {
92 ref WagonSlotComponent wagonSlot = ref world.Get<WagonSlotComponent>(wagon);
93 Vector3 wagonPosition = trainRoot.GlobalPosition + new Vector3(-wagonSlot.SlotIndex * wagonSize, 2.0f, 0);
94 float distanceToWagon = positionComponent.Value.DistanceTo(wagonPosition);
95 if (!(distanceToWagon < wagonPhysicalRadius + 0.5f)) { continue; }
96
97 Vector3 pushDirection = (positionComponent.Value - wagonPosition).Normalized();
98 repulsionForce += pushDirection * ((wagonPhysicalRadius + 0.5f) - distanceToWagon) * 5.0f;
99 neighbors++;
100 }
101
102 if (neighbors > 0) {
103 repulsionForce /= neighbors;
104 moveDirection = (moveDirection + repulsionForce * 1.5f).Normalized();
105 }
106
107 positionComponent.Value += moveDirection * movementComponent.Speed * (float)delta;
108 if (movementComponent.IsFlying) { continue; }
109
110 positionComponent.Value.Y -= 9.81f * (float)delta;
111 if (positionComponent.Value.Y < 0) {
112 positionComponent.Value.Y = 0;
113 }
114 }
115 }
116 }

125 private void HandleWagonDestruction(World world, Entity target) {
126 ref WagonSlotComponent hitSlot = ref world.Get<WagonSlotComponent>(target);
127 GD.Print($"Wagon at index {hitSlot.SlotIndex} (layer {hitSlot.Layer}) destroyed!");
128
129 if (hitSlot.SlotIndex == 0) { GD.PrintErr("!!! GAME OVER - LOCOMOTIVE DESTROYED !!!"); }
130
131 List<Entity> allWagonEntities = [.. world.Query<WagonSlotComponent>()];
132 foreach (Entity wagonEntity in allWagonEntities) {
133 if (!world.IsAlive(wagonEntity)) { continue; }
134
135 ref WagonSlotComponent wagonSlot = ref world.Get<WagonSlotComponent>(wagonEntity);
136 if (wagonSlot.SlotIndex == hitSlot.SlotIndex && wagonSlot.Layer >= hitSlot.Layer) { DestroyWagon(world, wagonEntity); }
137 else if (wagonSlot.SlotIndex > hitSlot.SlotIndex) { AbandonWagon(world, wagonEntity, trainRoot); }
138 }

160 private static void AbandonWagon(World world, Entity entity, Node3D nodeRoot) {

162 Node3D node = world.Get<RenderableComponent>(entity).Node;

164 Transform3D globalTransform = node.GlobalTransform;

169 timer.Timeout += () => {
170 if (GodotObject.IsInstanceValid(node)) { node.QueueFree(); }
171 };
172 }
173 }

186 private static Option<Entity> FindBestTarget(EnemyType enemyType, List<Entity> wagons, World world) {
187 Option<Entity> bestTarget = Option<Entity>.None;
188 float bestScore = -99999f;
189
190 foreach (Entity wagon in wagons) {
191 ref WagonTypeComponent wagonTypeComponent = ref world.Get<WagonTypeComponent>(wagon);
192 ref WagonSlotComponent wagonSlotComponent = ref world.Get<WagonSlotComponent>(wagon);
193 float score = GetDefaultTypePriority(wagonTypeComponent.Type) + (wagonSlotComponent.Layer * 50);
194 switch (enemyType) {
195 case EnemyType.Safeguard:
196 score += wagonSlotComponent.Layer * 10000;
197 break;
198 case EnemyType.Wasp:
199 switch (wagonTypeComponent.Type) {
200 case WagonType.Combat:
201 score += 5000;
202 break;
203 case WagonType.Locomotive:
204 case WagonType.Living:
205 case WagonType.Storage:
206 case WagonType.Research:
207 case WagonType.Medical:
208 break;
209 default:
210 throw new ArgumentOutOfRangeException();
211 }
212
213 break;
214 case EnemyType.Crawler:
215 switch (wagonTypeComponent.Type) {
216 case WagonType.Living:
217 score += 5000;
218 break;
219 case WagonType.Research:
220 score += 4000;
221 break;
222 case WagonType.Locomotive:
223 case WagonType.Combat:
224 case WagonType.Storage:
225 case WagonType.Medical:
226 break;
227 default:
228 throw new ArgumentOutOfRangeException();
229 }
230 break;
231 default:
232 throw new ArgumentOutOfRangeException(nameof(enemyType), enemyType, null);
233 }
234
235 if (!(score > bestScore)) { continue; }
236
237 bestScore = score;
238 bestTarget = Option<Entity>.Some(wagon);
239 }
240
241 return bestTarget;
242 }

265 private void SpawnHorde(World world, int count, EnemyType type) {
266 EnemyDefinition enemyDefinition = EnemyRegistry.EnemyDefs[type];
267 float angle = GD.Randf() * Mathf.Tau;
268 const float distance = 80f;
269 Vector3 hordeEpicenter = trainRoot.GlobalPosition +
270 new Vector3(Mathf.Cos(angle) * distance, 0, Mathf.Sin(angle) * distance);
271
272 for (int enemyIndex = 0; enemyIndex < count; enemyIndex++) {
273 Entity enemyEntity = world.CreateEntity();
274
275 Vector3 randomOffset = new(
276 (GD.Randf() - 0.5f) * enemyDefinition.DispersionRadius * 2f,
277 enemyDefinition.IsFlying ? 5.0f : 0f,
278 (GD.Randf() - 0.5f) * enemyDefinition.DispersionRadius * 2f
279 );
280
281 world.Add(enemyEntity, new PositionComponent { Value = hordeEpicenter + randomOffset });
282 world.Add(enemyEntity, new MovementComponent { Speed = enemyDefinition.Speed + (GD.Randf() * 2f), IsFlying = enemyDefinition.IsFlying });
283 world.Add(enemyEntity, new HealthComponent { Max = enemyDefinition.Health, Current = enemyDefinition.Health });
284 world.Add(
285 enemyEntity,
286 new EnemyComponent {
287 Type = enemyDefinition.Type,
288 Damage = enemyDefinition.Damage,
289 AttackRange = enemyDefinition.AttackRange,
290 AttackSpeed = enemyDefinition.AttackSpeed
291 }
292 );
293 }
