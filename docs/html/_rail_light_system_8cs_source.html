76 private static Vector3 CalculatePositionOnRail(ref MapComponent map, ref LocationComponent location, float lookAhead) {
77 MapNode startNode = map.AllNodes[location.CurrentNodeId];
78 MapNode targetNode = map.AllNodes[location.TargetNodeId];
79 Vector3 startPosition = new(startNode.Position.X, 0, startNode.Position.Y);
80 Vector3 endPosition = new(targetNode.Position.X, 0, targetNode.Position.Y);
81 float segmentDistance = startPosition.DistanceTo(endPosition);
82 float totalProgress = location.TravelProgress + lookAhead;
83 if (totalProgress <= segmentDistance) {
84 float interpolationFactor = totalProgress / segmentDistance;
85 return RailSampler.SampleBezier(startPosition, endPosition, location.CurrentNodeId, location.TargetNodeId, interpolationFactor) + Vector3.Up * 6f;
86 }
87
88 if (targetNode.NextNodes.Count <= 0) { return endPosition + Vector3.Up * 6f; }
89
90 MapNode nextNode = map.AllNodes[targetNode.NextNodes[0]];
91 Vector3 nextPosition = new(nextNode.Position.X, 0, nextNode.Position.Y);
92 float overflow = totalProgress - segmentDistance;
93 float nextSegmentDistance = endPosition.DistanceTo(nextPosition);
94 float nextInterpolationFactor = Mathf.Clamp(overflow / nextSegmentDistance, 0, 1);
95 return RailSampler.SampleBezier(endPosition, nextPosition, targetNode.Id, nextNode.Id, nextInterpolationFactor) + Vector3.Up * 6f;
