113 Vector2 centerOffset = GetRect().Size / 2f;

116 MapNode node = map.AllNodes[nextNodeId];

213 private static Vector2 GetTrainMapPosition(MapComponent map, LocationComponent location) {
214 Vector2 start = map.AllNodes[location.CurrentNodeId].Position;
215 Vector2 end = map.AllNodes[location.TargetNodeId].Position;
216
217 Vector3 startPosition3D = new(start.X, 0, start.Y);
218 Vector3 endPosition3D = new(end.X, 0, end.Y);
219
220 float segmentDistance = startPosition3D.DistanceTo(endPosition3D);
221 float interpolationFactor = segmentDistance > 0 ? Mathf.Clamp(location.TravelProgress / segmentDistance, 0, 1) : 0;
222
223 Vector3 position3D = RailSampler.SampleBezier(startPosition3D, endPosition3D, location.CurrentNodeId, location.TargetNodeId, interpolationFactor);
224 return new Vector2(position3D.X, position3D.Z);
225 }
