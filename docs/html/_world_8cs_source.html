721public class ComponentStore<T> : IComponentStore where T : struct {
727 private T[] _instances = new T[1024];
728
736 private int[] _sparse = new int[4096];
737
743 private int[] _dense = new int[1024];
744
750 public ComponentStore() {
751 Array.Fill(_sparse, -1);
752 }
753
760 public void Add(int entityId, T component) {
761 if (entityId >= _sparse.Length) {
762 Array.Resize(ref _sparse, entityId * 2);
763 }
764
765 if (Count >= _instances.Length) {
766 Array.Resize(ref _instances, Count * 2);
767 Array.Resize(ref _dense, Count * 2);
768 }
769
770 int index = Count++;
771 _instances[index] = component;
772 _dense[index] = entityId;
773 _sparse[entityId] = index;
774 }
775
785 public ref T Get(int entityId) => ref _instances[_sparse[entityId]];
786
793 public bool Has(int entityId) => entityId < _sparse.Length && _sparse[entityId] != -1;
794
800 public void Remove(int entityId) {
801 if (!Has(entityId)) {
802 return;
803 }
804
805 int targetIndex = _sparse[entityId];
806 _instances[targetIndex] = _instances[--Count];
807 _dense[targetIndex] = _dense[Count];
808 _sparse[_dense[targetIndex]] = targetIndex;
809 _sparse[entityId] = -1;
810 }
811
816 public int Count { get; private set; }
817
823 public ref T GetByIndex(int index) => ref _instances[index];
