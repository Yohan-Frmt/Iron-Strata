276 private void DrawCard() {
277 Option<Entity> resourceEntityOption = _world.QueryFirst<ResourceComponent>();
278 if (resourceEntityOption.IsNone) { return; }
279
280 ref ResourceComponent resources = ref _world.Get<ResourceComponent>(resourceEntityOption.Unwrap());
281 if (resources.Scrap < ResourceRegistry.CardDrawCost || _handContainer.GetChildCount() >= 5) { return; }
282
283 resources.Scrap -= ResourceRegistry.CardDrawCost;
284 CardUi newCard = _cardScene.Instantiate<CardUi>();
285 WagonType randomType = GD.Randf() > 0.5f ? WagonType.Combat : WagonType.Storage;
286 _handContainer.AddChild(newCard);
287 newCard.Setup(randomType);
288 }

359 private static Entity CreateWagon(
360 World world, int slot, int layer, WagonType type, string blueprint, float health,
361 Color tint, string label
362 ) {
363 Entity entity = world.CreateEntity();
364 world.Add(entity, new WagonTypeComponent { Type = type, BlueprintId = blueprint });
365 world.Add(entity, new WagonSlotComponent { SlotIndex = slot, Layer = layer });
366 world.Add(entity, new HealthComponent { Max = health, Current = health });
367 world.Add(entity, new RenderableComponent { Tint = tint, Label = label });
368
369 if (type == WagonType.Combat) {
370 world.Add(entity, new TurretComponent { Range = 30f, Damage = 15f, FireRate = 10f });
371 }
372
373 return entity;
